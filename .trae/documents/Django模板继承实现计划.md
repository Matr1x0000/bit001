```
static/
├── css/
│   ├── base.css          # 基础样式
│   ├── dashboard.css     # 仪表盘样式

│   ├── families.css      # 家庭档案样式

│   ├── families.js       # 家庭档案交互
│   └── charts.js         # 图表初始化
└── images/               # 图片资源
```
